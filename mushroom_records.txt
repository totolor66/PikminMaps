def main():

    if len(sys.argv) < 2:

        print("\n❌ 錯誤：缺少參數！")

        print("用法範例: python map_run.py Peru_Iquitos")

        sys.exit(1)



    location_name = sys.argv[1]



    if location_name not in LOCATIONS:

        print(f"❌ 錯誤：找不到地點 '{location_name}'")

        sys.exit(1)



    target_gps, grid_param = LOCATIONS[location_name]



    match = re.match(r"(\d+)x(\d+)", grid_param)

    if not match:

        print(f"❌ 錯誤：地點 '{location_name}' 的格線格式錯誤 '{grid_param}'")

        sys.exit(1)

    

    cols, rows = int(match.group(1)), int(match.group(2))



    print(f"\n🚀 任務啟動：{location_name} ({target_gps}) | 模式：{cols}x{rows}")



    # --- 階段 00：檢查螢幕狀態 ---

    ensure_screen_unlocked()  # ⭐️ 新增這行



    # --- 階段 0：檢查 APP 是否開啟 ---

    ensure_app_running()



    # --- 階段 1：GPS Joystick 自動定位 ---

    print("\n[階段 1] GPS 定位中...")

    adb_tap(UI_GPS_JOY_MENU)

    adb_tap(UI_GPS_JOY_TELEPORT_BTN)

    adb_tap(UI_GPS_JOY_INPUT_FIELD)

    adb_input_text(target_gps)

    adb_tap(UI_GPS_JOY_CONFIRM_BTN)

    adb_tap(UI_GPS_JOY_MENU)

    time.sleep(WAIT_GPS)



    # --- 階段 2：進入地圖模式與視野縮放 ---

    print("\n[階段 2] 準備地圖視野...")

    adb_tap(UI_GAME_MAP_BTN)

    adb_tap(UI_GAME_BIRD_VIEW_BTN)

    print("\n[階段 2.2] 點羅盤確認地圖朝北...")

    adb_tap(UI_GAME_MAP_COMPASS)

    time.sleep(WAIT_MAP_LOAD)

    adb_zoom_out()

    # 縮小兩次,確保有縮掉最小

    time.sleep(1)

    adb_zoom_out()



    # --- 階段 3：執行掃描 ---

    capture_custom_grid(location_name, cols, rows, target_gps)



    print(f"\n✅ 任務完成！共存檔 {cols*rows} 張圖片於當前目錄。")

    adb_tap(UI_GAME_MAP_EXIT)



if __name__ == "__main__":
